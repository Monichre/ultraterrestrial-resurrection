export async function getSightingById(id: string): Promise<SightingsRecord | null> {

}

export async function getAllSightings(options?: {
  filter?: Record<string, any>;
  sort?: { column: string; direction: "asc" | "desc" }[];
  page?: number;
  size?: number;
}): Promise<{
  records: SightingsRecord[];
  pagination: { hasNextPage: boolean; total?: number };
}> {
  const { filter, sort, page = 1, size = 50 } = options || {};
  let query = xata.db.sightings.filter(filter || {});

  if (sort?.length) {
    for (const { column, direction } of sort) {
      query = query.sort(column, direction);
    }
  }

  const result = await query.getPaginated({
    pagination: { size, offset: (page - 1) * size },
  });

  return {
    records: result.records,
    pagination: {
      hasNextPage: result.hasNextPage,
      total: result.pagination?.total,
    },
  };
}

import { xata } from '@/xata/client';

export interface DatabaseTableCursorRule {
  table: string;
  operation: 'create' | 'read' | 'update' | 'delete' | 'list' | 'search';
  filters?: Record<string, any>;
  sort?: { column: string; direction: 'asc' | 'desc' }[];
  pagination?: { size: number; offset: number };
}

export async function executeDatabaseOperation<T = any>(rule: DatabaseTableCursorRule): Promise<T> {
  const { table, operation, filters, sort, pagination } = rule;
  
  let query = (xata.db as any)[table];
  
  switch (operation) {
    case 'list':
      if (filters) query = query.filter(filters);
      if (sort?.length) {
        for (const { column, direction } of sort) {
          query = query.sort(column, direction);
        }
      }
      if (pagination) {
        return await query.getPaginated({ pagination });
      }
      return await query.getAll();
      
    case 'search':
      if (!filters?.query) throw new Error('Search query required');
      return await xata.search.all(filters.query, {
        tables: [{ table, target: filters.targets || [] }],
        fuzziness: filters.fuzziness || 1,
        prefix: filters.prefix || 'phrase',
        limit: filters.limit || 10,
      });
      
    case 'read':
      if (!filters?.id) throw new Error('ID required for read operation');
      return await query.read(filters.id);
      
    case 'create':
      if (!filters?.data) throw new Error('Data required for create operation');
      return await query.create(filters.data);
      
    case 'update':
      if (!filters?.id || !filters?.data) throw new Error('ID and data required for update operation');
      return await query.update(filters.id, filters.data);
      
    case 'delete':
      if (!filters?.id) throw new Error('ID required for delete operation');
      return await query.delete(filters.id);
      
    default:
      throw new Error(`Unsupported operation: ${operation}`);
  }
}

export async function initiateDatabaseTableQuery(params: {
  type: string;
  searchTerm?: string;
  limit?: number;
}) {
  const { type, searchTerm, limit = 10 } = params;
  
  if (searchTerm) {
    return await executeDatabaseOperation({
      table: type,
      operation: 'search',
      filters: {
        query: searchTerm,
        limit,
        targets: ['description', 'name', 'title', 'content'],
      },
    });
  }
  
  return await executeDatabaseOperation({
    table: type,
    operation: 'list',
    pagination: { size: limit, offset: 0 },
  });
}

export async function xataToXYFlow(data: any[], layout: string = 'horizontal') {
  // Transform database records into XYFlow nodes and edges
  const nodes = data.map((record, index) => ({
    id: record.id || `node-${index}`,
    type: 'default',
    position: calculatePosition(index, layout),
    data: {
      label: record.name || record.title || record.description?.substring(0, 50) || 'Unknown',
      record,
    },
  }));

  const edges = nodes.slice(1).map((node, index) => ({
    id: `edge-${index}`,
    source: nodes[0].id,
    target: node.id,
    type: 'default',
  }));

  return { nodes, edges };
}

function calculatePosition(index: number, layout: string) {
  switch (layout) {
    case 'horizontal':
      return { x: index * 200, y: 0 };
    case 'vertical':
      return { x: 0, y: index * 100 };
    case 'radial':
      const angle = (index * 2 * Math.PI) / 8;
      const radius = 200;
      return {
        x: Math.cos(angle) * radius,
        y: Math.sin(angle) * radius,
      };
    case 'grid':
      const cols = 3;
      return {
        x: (index % cols) * 200,
        y: Math.floor(index / cols) * 100,
      };
    default:
      return { x: index * 200, y: 0 };
  }
}

'use client';

import React, { createContext, useContext, useState, useCallback } from 'react';
import { Node, Edge, addEdge, Connection, useNodesState, useEdgesState } from '@xyflow/react';

export interface MindMapNode extends Node {
  parentId?: string;
  entityType?: string;
  searchTerm?: string;
  status?: 'loading' | 'success' | 'error';
  error?: string;
}

export interface MindMapContextType {
  nodes: MindMapNode[];
  edges: Edge[];
  setNodes: (nodes: MindMapNode[] | ((nodes: MindMapNode[]) => MindMapNode[])) => void;
  setEdges: (edges: Edge[] | ((edges: Edge[]) => Edge[])) => void;
  onNodesChange: (changes: any[]) => void;
  onEdgesChange: (changes: any[]) => void;
  onConnect: (connection: Connection) => void;
  addNode: (node: Omit<MindMapNode, 'id'> & { id?: string }) => string;
  addEdge: (edge: Omit<Edge, 'id'> & { id?: string }) => void;
  updateNode: (id: string, updates: Partial<MindMapNode>) => void;
  removeNode: (id: string) => void;
  clearGraph: () => void;
  findNodesByType: (type: string) => MindMapNode[];
  findNodesBySearchTerm: (searchTerm: string) => MindMapNode[];
}

const MindMapContext = createContext<MindMapContextType | null>(null);

export function MindMapProvider({ children }: { children: React.ReactNode }) {
  const [nodes, setNodes, onNodesChange] = useNodesState<MindMapNode>([]);
  const [edges, setEdges, onEdgesChange] = useEdgesState([]);

  const onConnect = useCallback(
    (params: Connection) => setEdges((eds) => addEdge(params, eds)),
    [setEdges]
  );

  const addNode = useCallback((nodeData: Omit<MindMapNode, 'id'> & { id?: string }) => {
    const id = nodeData.id || `node-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
    const newNode: MindMapNode = {
      ...nodeData,
      id,
      position: nodeData.position || { x: 0, y: 0 },
      data: nodeData.data || {},
    };
    
    setNodes((nds) => [...nds, newNode]);
    return id;
  }, [setNodes]);

  const addEdgeCallback = useCallback((edgeData: Omit<Edge, 'id'> & { id?: string }) => {
    const id = edgeData.id || `edge-${edgeData.source}-${edgeData.target}`;
    const newEdge: Edge = { ...edgeData, id };
    setEdges((eds) => [...eds, newEdge]);
  }, [setEdges]);

  const updateNode = useCallback((id: string, updates: Partial<MindMapNode>) => {
    setNodes((nds) => nds.map((node) => 
      node.id === id ? { ...node, ...updates } : node
    ));
  }, [setNodes]);

  const removeNode = useCallback((id: string) => {
    setNodes((nds) => nds.filter((node) => node.id !== id));
    setEdges((eds) => eds.filter((edge) => edge.source !== id && edge.target !== id));
  }, [setNodes, setEdges]);

  const clearGraph = useCallback(() => {
    setNodes([]);
    setEdges([]);
  }, [setNodes, setEdges]);

  const findNodesByType = useCallback((type: string) => {
    return nodes.filter((node) => node.type === type);
  }, [nodes]);

  const findNodesBySearchTerm = useCallback((searchTerm: string) => {
    return nodes.filter((node) => node.searchTerm === searchTerm);
  }, [nodes]);

  const value: MindMapContextType = {
    nodes,
    edges,
    setNodes,
    setEdges,
    onNodesChange,
    onEdgesChange,
    onConnect,
    addNode,
    addEdge: addEdgeCallback,
    updateNode,
    removeNode,
    clearGraph,
    findNodesByType,
    findNodesBySearchTerm,
  };

  return (
    <MindMapContext.Provider value={value}>
      {children}
    </MindMapContext.Provider>
  );
}

export function useMindMap() {
  const context = useContext(MindMapContext);
  if (!context) {
    throw new Error('useMindMap must be used within a MindMapProvider');
  }
  return context;
}

export interface SessionNote {
  id: string;
  content: string;

  type: 'user' | 'assistant' | 'system';
}

export interface SessionNotesContextType {
  notes: SessionNote[];
  addNote: (content: string, type: SessionNote['type']) => void;
  removeNote: (id: string) => void;
  clearNotes: () => void;
  saveMessageAsNote: (message: { content: string; role: string }) => void;
}

const SessionNotesContext = createContext<SessionNotesContextType | null>(null);

export function SessionNotesProvider({ children }: { children: React.ReactNode }) {
  const [notes, setNotes] = useState<SessionNote[]>([]);

  const addNote = useCallback((content: string, type: SessionNote['type']) => {
    const newNote: SessionNote = {
      id: `note-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
      content,
      timestamp: new Date(),
      type,
    };
    setNotes((prev) => [...prev, newNote]);
  }, []);

  const removeNote = useCallback((id: string) => {
    setNotes((prev) => prev.filter((note) => note.id !== id));
  }, []);

  const clearNotes = useCallback(() => {
    setNotes([]);
  }, []);

  const saveMessageAsNote = useCallback((message: { content: string; role: string }) => {
    addNote(message.content, message.role as SessionNote['type']);
  }, [addNote]);

  const value: SessionNotesContextType = {
    notes,
    addNote,
    removeNote,
    clearNotes,
    saveMessageAsNote,
  };

  return (
    <SessionNotesContext.Provider value={value}>
      {children}
    </SessionNotesContext.Provider>
  );
}

          status: 'success',
          data: { label: `${type} (${newNodes.length} records)` },
        });
      } else {
        mindMap.updateNode(userNodeId, { 
          status: 'error',
          error: 'No data found',
        });
      }
    } catch (error) {
      console.error('Data loading error:', error);
      mindMap.updateNode(userNodeId, { 
        status: 'error',
        error: error instanceof Error ? error.message : 'Failed to load data',
      });
    }
  }, [mindMap, generateNodeId, calculateCenterOfScreen, computeChildPositions]);

  const addDataToMindMap = useCallback((type: string) => {
    handleLoadingRecords({ data: { type } });
  }, [handleLoadingRecords]);

  // Event handlers
  const handleOracleAction = useCallback(async () => {
    const inputValue = localInput.trim();
    
    if (!inputValue) return;

    if (activeCommand === 'chat' || (!activeCommand && !selectedModel)) {
      // Handle chat
      handleInputChange({ target: { value: inputValue } } as any);
      await submitMessage();
      setLocalInput('');
    } else if (activeCommand === 'search' || selectedModel) {
      // Handle search
      const type = selectedModel || 'sightings';
      await runSearch({ type, searchTerm: inputValue });
      setLocalInput('');
    } else if (selectedModel && !inputValue) {
      // Load model data
      addDataToMindMap(selectedModel);
    }

    setActiveCommand('');
  }, [localInput, activeCommand, selectedModel, handleInputChange, submitMessage, runSearch, addDataToMindMap]);

  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      handleOracleAction();
    } else if (e.key === 'Backspace' && localInput === '' && activeCommand) {
      setActiveCommand('');
    } else if (e.key === '/' && !isCommandMenuOpen) {
      e.preventDefault();
      setIsCommandMenuOpen(true);
    } else if (!activeCommand && !selectedModel && localInput) {
      // Default to search
      setActiveCommand('search');
    }
  }, [localInput, activeCommand, isCommandMenuOpen, handleOracleAction]);

  const handleChange = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
    const value = e.target.value;
    setLocalInput(value);
    
    if (activeCommand === 'chat') {
      handleInputChange(e);
    }
  }, [activeCommand, handleInputChange]);

  const handleCommandSelect = useCallback((commandId: string) => {
    setActiveCommand(commandId);
    setIsCommandMenuOpen(false);
    inputRef.current?.focus();
  }, []);

  const handleFormSubmit = useCallback((e: React.FormEvent) => {
    e.preventDefault();
    if (activeCommand === 'chat') {
      submitMessage();
      setLocalInput('');
    } else {
      handleOracleAction();
    }
  }, [activeCommand, submitMessage, handleOracleAction]);

  const isLoading = status === 'in_progress';

  return (
    <div className="fixed bottom-0 left-0 right-0 bg-white border-t border-gray-200 p-4 shadow-lg">
      <div className="max-w-4xl mx-auto space-y-4">
        {/* Error display */}
        {error && (
          <div className="bg-red-50 border border-red-200 text-red-700 px-4 py-2 rounded-lg">
            {error}
          </div>
        )}

        {/* Messages */}
        {messages.length > 0 && (
          <MessagesWithNotesSaving messages={messages} />
        )}

        {/* Main input area */}
        <div className="flex items-center space-x-4">
          {/* Model selection */}
          <div className="w-48">
            <UltraterrestrialModelSelection
              value={selectedModel}
              onValueChange={setSelectedModel}
              models={MODELS}
              placeholder="Select model..."
            />
          </div>

          {/* Input with command menu */}
          <div className="flex-1 relative">
            <OracleCommandMenu
              isOpen={isCommandMenuOpen}
              onClose={() => setIsCommandMenuOpen(false)}
              onSelect={handleCommandSelect}
              commands={COMMANDS}
            />
            
            <form onSubmit={handleFormSubmit}>
              <OracleInput
                ref={inputRef}
                value={localInput}
                onChange={handleChange}
                onKeyDown={handleKeyDown}
                placeholder={
                  activeCommand
                    ? `${COMMANDS.find(c => c.id === activeCommand)?.label}...`
                    : selectedModel
                    ? `Search ${MODELS.find(m => m.id === selectedModel)?.label}...`
                    : "Ask Oracle... (press / for commands)"
                }
                isLoading={isLoading}
                command={activeCommand}
                model={selectedModel}
              />
            </form>
          </div>

          {/* Submit button */}
          <button
            onClick={handleOracleAction}
            disabled={isLoading || !localInput.trim()}
            className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed"
          >
            {isLoading ? 'Processing...' : 'Submit'}
          </button>
        </div>

        {/* Status indicators */}
        <div className="flex items-center justify-between text-sm text-gray-500">
          <div>
            Session: {sessionId.substring(0, 8)}...
          </div>
          <div className="flex items-center space-x-4">
            <span>Nodes: {mindMap.nodes.length}</span>
            <span>Messages: {messages.length}</span>
            {isLoading && <span className="text-blue-600">AI thinking...</span>}
          </div>
        </div>
      </div>
    </div>
  );
}
