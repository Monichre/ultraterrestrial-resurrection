# Timeline Scrubber & Network Explorer - Agentic Architecture Implementation

**Created**: 2025-11-25  
**Status**: Revised Agentic Architecture  
**Priority**: Critical - Proper AI Integration

## Executive Summary

This document provides the **REVISED** implementation plan for Timeline Scrubber and Network Explorer, properly integrated with the **Prometheus AI agentic layer**. All graph nodes and UI rendering are functions of agent analysis, not direct database queries.

### Critical Architectural Principle

> **"All graph nodes and UI should be rendered as a function of the agentic layer"**

This means:
- ❌ NO direct database queries for graph construction
- ❌ NO hardcoded templates or manual graph building
- ✅ Prometheus AI orchestrates all operations
- ✅ Agent tools analyze and return graph structures
- ✅ Contextual Intelligence enriches all outputs
- ✅ Enhanced Nodes render agent results

```typescript
// apps/app/src/features/mindmap/agents/temporal-navigation-agent.ts

import { TemporalNavigationResult, EraStatisticsResult } from '../tools/ultraterrestrial-agent-tools'
import { askXataWithAi } from '@db/xata'
import { enhanceLayoutWithAI } from '../actions/ai-actions'
import { getGraphContext } from '../utils/contextual-intelligence'

/**
 * Temporal Navigation Agent
 * Uses Prometheus AI to analyze historical context and generate temporal visualizations
 */
export async function temporalNavigationAgent(
  year: number,
  context?: {
    focusArea?: 'global' | 'regional' | 'thematic'
    includeKeyEventsOnly?: boolean
    maxResults?: number
  }
): Promise<TemporalNavigationResult> {
  try {
    // 1. Ask Prometheus AI to analyze temporal context
    const prompt = `
      Analyze the historical context for year ${year} in UFO/UAP disclosure history.
      
      Provide:
      1. Era classification and significance
      2. Key events within ±2 years
      3. Historical themes and patterns
      4. Geographic distribution of activity
      5. Notable personnel and organizations
      6. Narrative progression context
      
      Focus: ${context?.focusArea || 'global'}
      Max events: ${context?.maxResults || 20}
    `
    
    // Use Xata AI search with Prometheus guidance
    const aiResponse = await askXataWithAi({
      question: prompt,
      table: 'events',
      rules: [
        `Focus on events between ${year - 2} and ${year + 2}`,
        'Identify historical significance and patterns',
        'Discover relationships between events',
        'Provide narrative context'
      ]
    })
    
    // 2. Extract events and entities from AI response
    const events = aiResponse.records
    const entities = extractEntitiesFromEvents(events)
    
    // 3. Generate AI-enhanced graph layout
    const layoutResult = await enhanceLayoutWithAI({
      entities: events,
      sourceNodeId: `year-${year}`,
      entityType: 'temporal-events'
    })
    
    // 4. Convert to graph structure
    const graphData = await convertToGraphStructure(
      events,
      layoutResult.positions,
      layoutResult.relationships
    )
    
    // 5. Apply contextual intelligence enrichment
    const graphContext = getGraphContext(graphData.nodes)
    const enrichedNodes = await enrichNodesWithContext(
      graphData.nodes,
      graphContext
    )
    
    // 6. Analyze era context
    const eraAnalysis = await analyzeEraContext(year, events)
    
    // 7. Generate navigation suggestions
    const navigationContext = await generateNavigationSuggestions(
      year,
      eraAnalysis,
      graphContext
    )
    
    return {
      success: true,
      message: `Temporal analysis complete for year ${year}`,
      temporalContext: {
        currentYear: year,
        currentDate: new Date(year, 0, 1).toISOString(),
        eraAnalysis,
        keyEvents: events.map(e => ({
          id: e.id,
          title: e.title,
          date: e.date,
          significance: calculateSignificance(e),
          context: e.description || '',
          connections: extractConnections(e, events)
        })),
        statistics: {
          totalEvents: events.length,
          significantEvents: events.filter(e => e.category?.includes('significant')).length,
          geographicDistribution: calculateGeographicDistribution(events),
          categoryBreakdown: calculateCategoryBreakdown(events)
        }
      },
      graphVisualization: {
        success: true,
        message: 'Graph generated',
        graphData: {
          nodes: enrichedNodes,
          edges: graphData.edges,
          layout: {
            type: 'horizontal',
            applied: true,
            nodePositions: layoutResult.positions
          }
        },
        transformationMeta: {
          sourceRecords: events.length,
          successfulTransformations: enrichedNodes.length,
          layoutOptimizations: layoutResult.insights?.map(i => i.text) || [],
          enhancementLevel: 0.85
        }
      },
      navigationContext,
      relationalContext: extractRelationalContext(events, entities),
      suggestedNextActions: [
        `Explore ${navigationContext.nextSignificantYear}`,
        'View key figures from this era',
        'Examine geographic patterns'
      ],
      performance: {
        duration: 0, // tracked by caller
        recordsProcessed: events.length,
        nodesCreated: enrichedNodes.length,
        edgesCreated: graphData.edges.length
      }
    }
  } catch (error) {
    console.error('Temporal navigation agent error:', error)
    return {
      success: false,
      message: 'Failed to analyze temporal context',
      error: error instanceof Error ? error.message : 'Unknown error',
      temporalContext: null as any,
      graphVisualization: null as any,
      navigationContext: null as any,
      relationalContext: null as any
    }
  }
}

// Helper functions
function extractEntitiesFromEvents(events: any[]) {
  // Extract personnel, organizations, topics from events
  return events.flatMap(e => [
    ...(e['event-subject-matter-experts'] || []),
    ...(e.organizations || []),
    ...(e.topics || [])
  ])
}

function calculateSignificance(event: any): number {
  // AI-driven significance scoring
  let score = 0
  if (event.category?.includes('key-event')) score += 0.5
  if (event.category?.includes('significant')) score += 0.3
  if (event['event-subject-matter-experts']?.length > 2) score += 0.2
  return Math.min(1, score)
}
```

```typescript
// apps/app/src/features/mindmap/components/timeline-scrubber/hooks/use-timeline-agent.ts

import { useState, useCallback } from 'react'
import { temporalNavigationAgent, eraStatisticsAgent } from '@/features/mindmap/agents/temporal-navigation-agent'
import { useMindMap } from '@/contexts/mindmap/mindmap-context'
import type { TemporalNavigationResult } from '@/features/mindmap/tools/ultraterrestrial-agent-tools'

/**
 * Timeline Agent Hook
 * Orchestrates temporal navigation through Prometheus AI
 */
export function useTimelineAgent() {
  const { setNodes, setEdges } = useMindMap()
  const [agentResult, setAgentResult] = useState<TemporalNavigationResult | null>(null)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<Error | null>(null)
  
  /**
   * Navigate to specific year using agent analysis
   */
  const navigateToYear = useCallback(async (
    year: number,
    options?: {
      focusArea?: 'global' | 'regional' | 'thematic'
      includeKeyEventsOnly?: boolean
    }
  ) => {
    setLoading(true)
    setError(null)
    
    try {
      const startTime = Date.now()
      
      // Call temporal navigation agent
      const result = await temporalNavigationAgent(year, options)
      
      if (!result.success) {
        throw new Error(result.error || 'Agent analysis failed')
      }
      
      // Update performance metrics
      result.performance!.duration = Date.now() - startTime
      
      // Extract graph visualization from agent result
      const { graphData } = result.graphVisualization
      
      // Update graph with agent-generated nodes and edges
      setNodes(graphData.nodes)
      setEdges(graphData.edges)
      
      // Store agent result for UI rendering
      setAgentResult(result)
      
      // Fit view to show all agent-generated nodes
      fitView({ padding: 0.1, duration: 800 })
      
    } catch (err) {
      setError(err as Error)
      console.error('Timeline agent error:', err)
    } finally {
      setLoading(false)
    }
  }, [setNodes, setEdges])
  
  /**
   * Load era statistics through agent
   */
  const loadEraStatistics = useCallback(async () => {
    setLoading(true)
    try {
      const result = await eraStatisticsAgent()
      return result
    } catch (err) {
      setError(err as Error)
      throw err
    } finally {
      setLoading(false)
    }
  }, [])
  
  return {
    agentResult,
    loading,
    error,
    navigateToYear,
    loadEraStatistics
  }
}
```

```typescript
'use client'

import { useState, useEffect } from 'react'
import { Slider } from '@/components/ui/slider'
import { Button } from '@/components/ui/button'
import { Play, Pause, ChevronLeft, ChevronRight } from 'lucide-react'
import { useTimelineAgent } from './hooks/use-timeline-agent'

export function TimelineSlider() {
  const [year, setYear] = useState(2004)
  const [isPlaying, setIsPlaying] = useState(false)
  const [speed, setSpeed] = useState(1)
  const { navigateToYear, agentResult, loading } = useTimelineAgent()
  
  // Agent-driven year navigation
  const handleYearChange = async (newYear: number) => {
    setYear(newYear)
    await navigateToYear(newYear)
  }
  
  // Playback control
  useEffect(() => {
    if (!isPlaying) return
    
    const interval = setInterval(() => {
      setYear(prev => {
        const next = prev + speed
        if (next > 2024) {
          setIsPlaying(false)
          return 2024
        }
        navigateToYear(next)
        return next
      })
    }, 1000)
    
    return () => clearInterval(interval)
  }, [isPlaying, speed, navigateToYear])
  
  return (
    <div className="w-full p-4 bg-neutral-800/90 rounded-lg">
      {/* Year Display with Agent Context */}
      <div className="text-center mb-4">
        <h3 className="text-2xl font-bold text-white">{year}</h3>
        {agentResult && (
          <p className="text-sm text-gray-400">
            {agentResult.temporalContext.eraAnalysis.name}
          </p>
        )}
      </div>
      
      {/* Timeline Slider */}
      <Slider
        value={[year]}
        onValueChange={([value]) => handleYearChange(value)}
        min={1600}
        max={2024}
        step={1}
        disabled={loading}
        className="mb-4"
      />
      
      {/* Playback Controls */}
      <div className="flex items-center justify-center gap-2">
        <Button
          size="icon"
          variant="ghost"
          onClick={() => handleYearChange(year - 1)}
          disabled={loading || year <= 1600}
        >
          <ChevronLeft className="h-4 w-4" />
        </Button>
        
        <Button
          size="icon"
          onClick={() => setIsPlaying(!isPlaying)}
          disabled={loading}
        >
          {isPlaying ? (
            <Pause className="h-4 w-4" />
          ) : (
            <Play className="h-4 w-4" />
          )}
        </Button>
        
        <Button
          size="icon"
          variant="ghost"
          onClick={() => handleYearChange(year + 1)}
          disabled={loading || year >= 2024}
        >
          <ChevronRight className="h-4 w-4" />
        </Button>
        
        <div className="ml-4 text-sm text-gray-400">
          Speed: {speed}x
        </div>
      </div>
      
      {/* Agent Insights */}
      {agentResult && (
        <div className="mt-4 p-3 bg-neutral-700/30 rounded">
          <p className="text-xs text-gray-400">
            {agentResult.temporalContext.statistics.totalEvents} events discovered
          </p>
          <p className="text-xs text-gray-400">
            {agentResult.temporalContext.keyEvents.length} significant events
          </p>
        </div>
      )}
    </div>
  )
}
```

```typescript
      question: prompt,

        'Generate network layout'
```

```typescript
/**
 * Network Agent Hook
 * Orchestrates network exploration through Prometheus AI
 */
export function useNetworkAgent() {
  const { setNodes, setEdges } = useMindMap()
  const [agentResult, setAgentResult] = useState<NetworkTemplateResult | null>(null)
  const [availableNetworks, setAvailableNetworks] = useState<NetworkDiscoveryResult | null>(null)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<Error | null>(null)
  
  /**
   * Discover available network templates on mount
   */
  useEffect(() => {
    discoverNetworks()
  }, [])
  
  /**
   * Discover available networks through agent
   */
  const discoverNetworks = useCallback(async () => {
    setLoading(true)
    try {
      const result = await networkDiscoveryAgent()
      setAvailableNetworks(result)
    } catch (err) {
      setError(err as Error)
    } finally {
      setLoading(false)
    }
  }, [])
  
  /**
   * Load network template using agent analysis
   */
  const loadNetworkTemplate = useCallback(async (
    templateId: string,
    options?: {
      maxDepth?: number
      centerNodeId?: string
    }
  ) => {
    setLoading(true)
    setError(null)
    
    try {
      const startTime = Date.now()
      
      // Find template config
      const template = availableNetworks?.discoveredNetworks.find(
        n => n.id === templateId
      )
      
      // Call network template agent
      const result = await networkTemplateAgent({
        templateId,
        networkType: template?.networkType,
        centerNodeId: options?.centerNodeId,
        maxDepth: options?.maxDepth || 3
      })
      
      if (!result.success) {
        throw new Error(result.error || 'Agent analysis failed')
      }
      
      // Update performance metrics
      result.performance!.duration = Date.now() - startTime
      
      // Extract graph visualization from agent result
      const { graphData } = result.graphVisualization
      
      // Update graph with agent-generated network
      setNodes(graphData.nodes)
      setEdges(graphData.edges)
      
      // Store agent result for UI rendering
      setAgentResult(result)
      
      // Fit view
      fitView({ padding: 0.1, duration: 800 })
      
    } catch (err) {
      setError(err as Error)
      console.error('Network agent error:', err)
    } finally {
      setLoading(false)
    }
  }, [availableNetworks, setNodes, setEdges])
  
  return {
    agentResult,
    availableNetworks,
    loading,
    error,
    loadNetworkTemplate,
    discoverNetworks
  }
}
```
